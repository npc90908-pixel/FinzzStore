// ============================================================================
// AUTH — server side sessions on top of Supabase Auth (or the dev provider).
//
// Flow: the browser only ever receives two cookies.
//   fz_session : opaque 32-byte token (HttpOnly) — the database stores its sha256
//   fz_csrf    : CSRF token (readable by JS) — the database stores its sha256
// The Supabase access/refresh tokens never leave the server; they are stored in
// the sessions table encrypted with AES-256-GCM (APP_ENCRYPTION_KEY).
// ============================================================================
import { AppError, errors } from './errors.js';
import { randomToken, sha256, encryptSecret, decryptSecret, timingSafeEqualStr } from './security.js';
import { cookies } from './http.js';

const USER_CACHE_TTL_MS = 60_000;

export function createAuthService({ db, config, logger }) {
  const userCache = new Map(); // sha256(access token) -> { user, expiresAt }
  const key = config.encryptionKey;

  const now = () => Date.now();

  const cacheGet = (accessToken) => {
    const entry = userCache.get(sha256(accessToken));
    if (!entry) return null;
    if (entry.expiresAt < now()) {
      userCache.delete(sha256(accessToken));
      return null;
    }
    return entry.user;
  };

  const readProfile = async (userId, email) => {
    let profile = await db.profiles.get(userId);
    if (!profile) {
      const created = await db.profiles.ensure({ userId, email, username: null });
      profile = created.profile;
    }
    return profile;
  };

  /**
   * Supabase Auth (GoTrue) failure -> HTTP response.
   *
   * Production 2026-10-26: `email rate limit exceeded` (provider code
   * over_email_send_rate_limit, HTTP 429) surfaced as `500 DB_ERROR` because
   * it was routed through the database error mapper. Provider rate limits are
   * a normal, expected condition and must reach the user as a clear 429 with a
   * wait time — never as an internal error, and never retried in a loop.
   */
  const AUTH_PROVIDER_MESSAGES = {
    OVER_EMAIL_SEND_RATE_LIMIT:
      'Batas pengiriman email dari penyedia tercapai. Silakan coba lagi dalam beberapa menit.',
    OVER_REQUEST_RATE_LIMIT:
      'Terlalu banyak permintaan ke layanan autentikasi. Silakan coba lagi sebentar lagi.',
    EMAIL_ADDRESS_INVALID: 'Alamat email tidak valid atau tidak dapat menerima email.',
    SIGNUP_DISABLED: 'Pendaftaran akun baru sedang dinonaktifkan.',
    EMAIL_EXISTS: 'Email ini sudah terdaftar. Silakan masuk atau reset password.',
    WEAK_PASSWORD: 'Password terlalu lemah. Gunakan minimal 8 karakter dengan huruf dan angka.',
    CAPTCHA_FAILED: 'Verifikasi keamanan gagal. Muat ulang halaman lalu coba lagi.'
  };

  const AUTH_PROVIDER_CODES = {
    over_email_send_rate_limit: 'OVER_EMAIL_SEND_RATE_LIMIT',
    email_rate_limit_exceeded: 'OVER_EMAIL_SEND_RATE_LIMIT',
    over_request_rate_limit: 'OVER_REQUEST_RATE_LIMIT',
    email_address_invalid: 'EMAIL_ADDRESS_INVALID',
    email_address_not_authorized: 'EMAIL_ADDRESS_INVALID',
    signup_disabled: 'SIGNUP_DISABLED',
    user_already_exists: 'EMAIL_EXISTS',
    email_exists: 'EMAIL_EXISTS',
    weak_password: 'WEAK_PASSWORD',
    captcha_failed: 'CAPTCHA_FAILED'
  };

  /** Seconds the caller should wait. Provider Retry-After wins, else a default. */
  const providerRetryAfter = (err, fallback) => {
    const raw = Number(err?.retryAfter);
    if (Number.isFinite(raw) && raw > 0) return Math.min(Math.ceil(raw), 3600);
    return fallback;
  };

  /**
   * @returns {AppError|null} a mapped error, or null when it is not a provider
   *   rate-limit/validation condition (the caller decides what to do then).
   */
  const mapAuthProviderError = (err, { fallbackRetryAfter = 300 } = {}) => {
    const rawCode = String(err?.supabaseCode || err?.code || '').toLowerCase();
    const message = String(err?.message || '');
    const isRateLimit =
      /rate limit/i.test(message) ||
      rawCode.includes('rate_limit') ||
      Number(err?.status) === 429;

    if (isRateLimit) {
      const retryAfter = providerRetryAfter(err, fallbackRetryAfter);
      return new AppError(
        'EMAIL_RATE_LIMITED',
        AUTH_PROVIDER_MESSAGES.OVER_EMAIL_SEND_RATE_LIMIT,
        429,
        { retryAfter, field: null }
      );
    }

    const code = AUTH_PROVIDER_CODES[rawCode];
    if (code && AUTH_PROVIDER_MESSAGES[code]) {
      const status = code === 'SIGNUP_DISABLED' ? 403 : code === 'EMAIL_EXISTS' ? 409 : 422;
      return new AppError(code, AUTH_PROVIDER_MESSAGES[code], status);
    }
    return null;
  };

  const buildSessionRecord = async ({ authSession, userId, req }) => {
    const token = randomToken(32);
    const csrfToken = randomToken(24);
    const expiresAt = new Date(now() + config.cookie.maxAgeSeconds * 1000).toISOString();
    await db.sessions.create({
      user_id: userId,
      token_hash: sha256(token),
      csrf_hash: sha256(csrfToken),
      refresh_token_enc: encryptSecret(authSession.refresh_token, key),
      access_token_enc: encryptSecret(authSession.access_token, key),
      access_expires_at: new Date((authSession.expires_at || Math.floor(now() / 1000) + 3600) * 1000).toISOString(),
      ip: req?.ip || null,
      user_agent: String(req?.headers?.['user-agent'] || '').slice(0, 300) || null,
      expires_at: expiresAt
    });
    return { sessionToken: token, csrfToken };
  };

  /** Refresh the Supabase access token when it is close to expiry. */
  const ensureFreshAccessToken = async (session) => {
    const expiresAt = session.access_expires_at ? new Date(session.access_expires_at).getTime() : 0;
    if (expiresAt > now() + 60_000 && session.access_token_enc) {
      return decryptSecret(session.access_token_enc, key);
    }
    const refreshToken = decryptSecret(session.refresh_token_enc, key);
    if (!refreshToken) throw errors.unauthorized();

    let refreshed;
    try {
      refreshed = await db.auth.refresh(refreshToken);
    } catch (err) {
      logger.info('session refresh failed; revoking', { sessionId: session.id, error: err?.message });
      await db.sessions.revoke(session.id);
      throw errors.unauthorized();
    }
    await db.sessions.update(session.id, {
      refresh_token_enc: encryptSecret(refreshed.refresh_token, key),
      access_token_enc: encryptSecret(refreshed.access_token, key),
      access_expires_at: new Date((refreshed.expires_at || Math.floor(now() / 1000) + 3600) * 1000).toISOString(),
      last_used_at: new Date().toISOString()
    });
    return refreshed.access_token;
  };

  const resolve = async (req) => {
    const jar = cookies(req);
    const raw = jar[config.cookie.name];
    // `req.sessionState` lets requireUser tell "never signed in" (401) apart
    // from "your session ended" (401 SESSION_EXPIRED) for a clearer UX.
    req.sessionState = 'none';
    if (!raw || raw.length > 200) return null;

    const session = await db.sessions.byTokenHash(sha256(raw));
    if (!session) {
      req.sessionState = 'expired';
      return null;
    }
    if (session.revoked_at) {
      req.sessionState = 'expired';
      return null;
    }
    if (new Date(session.expires_at).getTime() < now()) {
      await db.sessions.revoke(session.id);
      req.sessionState = 'expired';
      return null;
    }

    let accessToken;
    try {
      accessToken = await ensureFreshAccessToken(session);
    } catch {
      return null;
    }

    let user = cacheGet(accessToken);
    if (!user) {
      user = await db.auth.getUser(accessToken);
      if (!user) {
        await db.sessions.revoke(session.id);
        return null;
      }
      userCache.set(sha256(accessToken), { user, expiresAt: now() + USER_CACHE_TTL_MS });
    }

    const profile = await readProfile(session.user_id, user.email);
    if (profile.id !== user.id) {
      logger.error('session user mismatch', { sessionId: session.id });
      await db.sessions.revoke(session.id);
      return null;
    }

    // Sliding "last used" update at most once every 5 minutes.
    const lastUsed = session.last_used_at ? new Date(session.last_used_at).getTime() : 0;
    if (now() - lastUsed > 5 * 60_000) {
      db.sessions.update(session.id, { last_used_at: new Date().toISOString() }).catch(() => {});
      db.profiles.touch(profile.id).catch(() => {});
    }

    return { session, user, profile, accessToken, csrfTokenHash: session.csrf_hash };
  };

  /**
   * Sliding session refresh: extends the cookie lifetime and rotates the CSRF
   * token. Called by the SPA when it detects an upcoming expiry, and after a
   * long idle period. The session row is reused (no new session for the same
   * device) so the session list stays meaningful.
   */
  const refreshSession = async (auth) => {
    const csrfToken = randomToken(24);
    const expiresAt = new Date(now() + config.cookie.maxAgeSeconds * 1000).toISOString();
    await db.sessions.update(auth.session.id, {
      csrf_hash: sha256(csrfToken),
      expires_at: expiresAt,
      last_used_at: new Date().toISOString()
    });
    return {
      sessionToken: null, // cookie value unchanged
      csrfToken,
      expiresAt,
      user: {
        id: auth.profile.id,
        email: auth.profile.email,
        username: auth.profile.username,
        role: auth.profile.role
      }
    };
  };

  const service = {
    /** Lazy per-request context: req.auth = { session, user, profile } | null */
    async attach(req) {
      if (req.auth !== undefined) return req.auth;
      try {
        req.auth = await resolve(req);
      } catch (err) {
        logger.warn('session resolution failed', { error: err?.message });
        req.auth = null;
      }
      return req.auth;
    },

    refresh: refreshSession,

    async requireUser(req) {
      const auth = await service.attach(req);
      if (!auth) {
        throw req.sessionState === 'expired' ? errors.sessionExpired() : errors.unauthorized();
      }
      if (auth.profile.status !== 'active') throw errors.accountSuspended();
      return auth;
    },

    async requireAdmin(req) {
      const auth = await service.requireUser(req);
      if (auth.profile.role !== 'admin') throw errors.forbidden();
      return auth;
    },

    async register({ email, password, username, req }) {
      const existing = await db.profiles.byEmail(email).catch(() => null);
      if (existing) {
        throw new AppError('EMAIL_TAKEN', 'Email ini sudah terdaftar. Silakan masuk atau reset password.', 409);
      }
      let created;
      try {
        created = await db.auth.signUp({
          email,
          password,
          metadata: { username: username || null }
        });
      } catch (err) {
        const mapped = mapAuthProviderError(err, { fallbackRetryAfter: 300 });
        if (mapped) {
          logger.warn('registration blocked by auth provider', {
            supabaseCode: err?.supabaseCode || null,
            providerStatus: err?.status || null,
            outcome: mapped.code
          });
          throw mapped;
        }
        logger.error('registration failed at auth provider', {
          supabaseCode: err?.supabaseCode || null,
          providerStatus: err?.status || null,
          error: err?.message
        });
        throw new AppError('REGISTRATION_FAILED', 'Pendaftaran gagal. Coba lagi beberapa saat lagi.', 503);
      }
      if (created.user?.id) {
        await db.profiles.ensure({ userId: created.user.id, email, username });
        if (created.user.email_confirmed_at) {
          await db.profiles.update(created.user.id, {}).catch(() => {});
        }
      }
      if (!created.session) {
        return { requiresVerification: true, email };
      }
      const authUser = created.user;
      const tokens = await buildSessionRecord({ authSession: created.session, userId: authUser.id, req });
      const profile = await readProfile(authUser.id, email);
      userCache.set(sha256(created.session.access_token), { user: authUser, expiresAt: now() + USER_CACHE_TTL_MS });
      return { requiresVerification: false, user: authUser, profile, ...tokens };
    },

    async login({ email, password, req }) {
      let authSession;
      try {
        authSession = await db.auth.signIn({ email, password });
      } catch (err) {
        const code = String(err?.supabaseCode || err?.code || '');
        if (code === 'email_not_confirmed') {
          throw new AppError('EMAIL_NOT_VERIFIED', 'Email belum diverifikasi. Periksa kotak masuk Anda.', 403);
        }
        const mapped = mapAuthProviderError(err, { fallbackRetryAfter: 60 });
        if (mapped) throw mapped;
        throw new AppError('INVALID_CREDENTIALS', 'Email atau password salah.', 401);
      }
      if (!authSession?.access_token) throw new AppError('LOGIN_FAILED', 'Login gagal. Coba lagi.', 401);

      const user = authSession.user || (await db.auth.getUser(authSession.access_token));
      if (!user) throw new AppError('LOGIN_FAILED', 'Login gagal. Coba lagi.', 401);

      const profile = await readProfile(user.id, user.email);
      if (profile.status !== 'active') throw errors.accountSuspended();

      const tokens = await buildSessionRecord({ authSession, userId: user.id, req });
      userCache.set(sha256(authSession.access_token), { user, expiresAt: now() + USER_CACHE_TTL_MS });
      return { user, profile, ...tokens };
    },

    async logout(req) {
      const auth = await service.attach(req);
      if (auth) {
        userCache.delete(sha256(auth.accessToken));
        await db.sessions.revoke(auth.session.id).catch(() => {});
        await db.auth.signOut(auth.accessToken).catch(() => {});
      }
    },

    async logoutAll(userId, exceptSessionId = null) {
      const { rows } = await db.sessions.listForUser?.(userId, { limit: 100 }) ?? { rows: [] };
      for (const s of rows) {
        if (s.id !== exceptSessionId) await db.sessions.revoke(s.id);
      }
      await db.profiles.touch(userId).catch(() => {});
      if (!rows.length) await db.sessions.revokeAllForUser(userId);
    },

    /** Step 2 of email verification: the client sends the token from the email link. */
    async verifyEmail({ accessToken, tokenHash, type = 'signup', req }) {
      let authSession = null;
      let user = null;

      if (accessToken) {
        user = await db.auth.getUser(accessToken);
        if (!user) throw new AppError('INVALID_TOKEN', 'Tautan verifikasi tidak valid atau kedaluwarsa.', 400);
        // A verified email link produces a valid session we can adopt server side.
        const refresh = req.body?.refresh_token;
        if (refresh) authSession = { access_token: accessToken, refresh_token: refresh, expires_at: Math.floor(now() / 1000) + 3600 };
      } else if (tokenHash) {
        authSession = await db.auth.verifyOtp({ tokenHash, type });
        user = authSession?.user;
      } else {
        throw errors.badRequest('Token verifikasi tidak ditemukan.');
      }

      if (!user) throw new AppError('INVALID_TOKEN', 'Tautan verifikasi tidak valid.', 400);
      const profile = await readProfile(user.id, user.email);
      await db.profiles.update(user.id, {}).catch(() => {});

      if (!authSession?.refresh_token) {
        return { verified: true, profile, requiresLogin: true };
      }
      const tokens = await buildSessionRecord({ authSession, userId: user.id, req });
      return { verified: true, user, profile, ...tokens, requiresLogin: false };
    },

    async requestPasswordReset(email) {
      await db.auth.sendPasswordReset(email);
      return { sent: true };
    },

    /**
     * Reset with a recovery token. In Supabase mode the frontend forwards the
     * access_token from the recovery link; GoTrue validates it and sets the new
     * password. In dev mode a one-time token is consumed.
     */
    async resetPassword({ accessToken, token, password }) {
      if (accessToken) {
        const user = await db.auth.getUser(accessToken);
        if (!user) throw new AppError('INVALID_TOKEN', 'Tautan reset tidak valid atau kedaluwarsa.', 400);
        await db.auth.updatePassword(accessToken, password);
        await db.sessions.revokeAllForUser(user.id).catch(() => {});
        return { ok: true };
      }
      if (token && typeof db.auth.consumeResetToken === 'function') {
        await db.auth.consumeResetToken(token, password);
        return { ok: true };
      }
      throw errors.badRequest('Token reset tidak ditemukan.');
    },

    async changePassword({ auth, currentPassword, newPassword }) {
      if (db.auth.kind === 'dev') {
        await db.auth.signIn({ email: auth.user.email, password: currentPassword });
        await db.auth.updatePassword(auth.accessToken, newPassword);
      } else {
        // Verify the current password through a fresh sign-in before changing it.
        try {
          await db.auth.signIn({ email: auth.user.email, password: currentPassword });
        } catch {
          throw new AppError('INVALID_CREDENTIALS', 'Password saat ini salah.', 401);
        }
        await db.auth.updatePassword(auth.accessToken, newPassword);
      }
      // Keep the current session, drop every other session.
      await db.sessions.revokeAllForUser(auth.user.id).catch(() => {});
      await db.sessions.update(auth.session.id, { revoked_at: null }).catch(() => {});
      return { ok: true };
    },

    async resendVerification(email) {
      await db.auth.resendVerification(email);
      return { sent: true };
    },

    /** Constant-time CSRF comparison against the stored hash. */
    verifyCsrf(req, auth) {
      const header = req.headers['x-csrf-token'];
      if (!header || !auth) return false;
      return timingSafeEqualStr(sha256(String(header)), auth.csrfTokenHash);
    },

    /** Periodic cleanup, called by the reconciliation job. */
    async cleanup() {
      await db.sessions.deleteExpired();
      userCache.clear();
    },

    _userCache: userCache
  };

  return service;
}
